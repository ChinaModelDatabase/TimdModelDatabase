## 1. 快速开始

### 1.1 下载安装

```bash
# 下载预编译二进制（国产平台）
wget https://github.com/tmDB/tmDB/releases/download/v1.0/tmdb-linux-arm64.tar.gz
tar -xzf tmdb-linux-arm64.tar.gz
cd tmdb

# 或从源码编译
git clone https://github.com/tmDB/tmDB.git
cd tmDB
cargo build --release
```

### 1.2 启动服务

> **⚠️ 当前不可用**：`start` 命令**不绑定任何端口**，仅打印日志并等待 Ctrl+C。
> `main.rs` 中该分支注释为 `TODO: bind listeners and accept connections`。
> 上述命令可运行，但不会接受连接。

```bash
# 默认配置启动
./tmdb start

# 指定配置文件
./tmdb start --config /etc/tmdb/tmdb.toml

# 指定监听地址
./tmdb start --listen 0.0.0.0:5432 --listen-mysql 0.0.0.0:3306
```

### 1.3 连接测试

```bash
# PostgreSQL 客户端
psql -h localhost -p 5432 -U tmdb

# MySQL 客户端
mysql -h localhost -P 3306 -u tmdb

# 命令行工具
./tmdb-cli -h localhost -p 5432
```

### 1.4 创建第一张表

```sql
-- PostgreSQL 方言
CREATE TABLE sensor_data (
    ts TIMESTAMPTZ NOT NULL,
    device_id BIGINT NOT NULL,
    temperature DOUBLE PRECISION,
    humidity DOUBLE PRECISION
);

-- MySQL 方言
CREATE TABLE sensor_data (
    ts DATETIME(3) NOT NULL,
    device_id BIGINT NOT NULL,
    temperature DOUBLE,
    humidity DOUBLE
);
```

### 1.5 写入数据

```sql
-- PostgreSQL 方言
INSERT INTO sensor_data VALUES
    (NOW(), 1001, 25.5, 60.2),
    (NOW(), 1002, 26.1, 58.8);

-- MySQL 方言
INSERT INTO sensor_data VALUES
    (NOW(3), 1001, 25.5, 60.2),
    (NOW(3), 1002, 26.1, 58.8);
```

### 1.6 查询数据

```sql
SELECT device_id, ts, temperature
FROM sensor_data
WHERE ts > NOW() - INTERVAL '1 hour'
ORDER BY ts DESC
LIMIT 10;
```

## 3. 配置说明

### 3.1 配置文件

> **⚠️ 下方配置项大多不存在**：`ServerConfig` 实际只有 3 个字段
> （`pg_listen`/`mysql_listen`/`data_dir`），`--config` 参数**从未被读取**。
> 下方 TOML 是目标形态；写入的 `[wire.*]`/`[storage]`/`[query]`/`[distributed]`/`[security]`
> 均不会生效。

配置文件位于 `/etc/tmdb/tmdb.toml`：

```toml
[server]
listen           = "0.0.0.0:5432"        # PostgreSQL 协议
listen_mysql     = "0.0.0.0:3306"        # MySQL 协议
listen_http      = "0.0.0.0:8080"        # HTTP 监控
max_connections  = 4096                   # 最大连接数
default_dialect  = "auto"                # 默认方言：auto/postgres/mysql

[wire.pg]
enabled          = true
auth_methods     = ["scram-sha-256", "md5", "sm3"]
max_bind_params  = 65535
copy_enabled     = true

[wire.mysql]
enabled                 = true
auth_plugins            = ["mysql_native_password", "sha256_password", "caching_sha2_password"]
default_charset         = "utf8mb4"
collation               = "utf8mb4_bin"
sql_mode                = "STRICT_TRANS_TABLES,NO_ENGINE_SUBSTITUTION"
autocommit_default      = true
max_allowed_packet      = "64MB"
prepared_stmt_enabled   = true

[storage]
data_dir        = "/data/tmdb"         # 数据目录
chunk_interval  = "1 day"                 # Chunk 间隔
segment_target  = "256MB"                 # 段文件目标大小
row_group_rows  = 4096                    # 每行组行数
page_size       = "64KB"                  # 页大小
default_encoding = "balanced"             # 编码档位：speed/balanced/archive

[query]
default_timeout        = "30s"
max_memory_per_query   = "2GB"
spill_to_disk          = true
plan_cache_size        = 1024

[distributed]
enabled        = true
shard_key      = "database"
replication    = 3
rebalance_rate = "50MB/s"

[security]
sm_tls        = true                      # 国密 TLS
sm_encrypt    = true                      # 静态 SM4 加密
audit_log     = "/var/log/tmdb/audit.log"
```

### 3.2 编码档位

| 档位 | 编码 | 压缩 | 适用场景 |
|---|---|---|---|
| `speed` | DeltaOfDelta | LZ4 | 高频写入、边端 |
| `balanced` | Gorilla | LZ4 | 通用（默认） |
| `archive` | Gorilla + 重编码 | zstd-3 | 长期留存 |

### 3.3 持久化档位

> **⚠️ 三档均不存在，且当前从不 fsync**：写入路径固定调用 `wal.append(..., durable=false)`，
> 无任何档位开关。进程崩溃可恢复（framed + CRC + 回放），**断电仍会丢数据**。
> 这是当前 P0 遗留项之一。

| 档位 | 行为 | 适用场景 |
|---|---|---|
| `max` | 每次提交 fsync | 金融、电网 SCADA |
| `normal` | 组提交 fsync（默认） | 通用 |
| `os_cache` | 不 fsync，靠 OS | 边端、可容忍丢失 |

## 4. SQL 语法

### 4.1 DDL

```sql
-- 创建数据库
CREATE DATABASE smart_grid;

-- 创建时序表（PostgreSQL 方言）
CREATE TABLE sensor_data (
    ts TIMESTAMPTZ NOT NULL,
    device_id BIGINT NOT NULL,
    temperature DOUBLE PRECISION,
    humidity DOUBLE PRECISION,
    PRIMARY KEY (ts, device_id)
);

-- 创建时序表（MySQL 方言）
CREATE TABLE sensor_data (
    ts DATETIME(3) NOT NULL,
    device_id BIGINT NOT NULL,
    temperature DOUBLE,
    humidity DOUBLE,
    PRIMARY KEY (ts, device_id)
) ENGINE=tmdb;

-- 创建索引
CREATE INDEX idx_device ON sensor_data (device_id);

-- 删除表
DROP TABLE sensor_data;
```

### 4.2 DML

```sql
-- 插入
INSERT INTO sensor_data VALUES (NOW(), 1001, 25.5, 60.2);

-- 批量插入
INSERT INTO sensor_data VALUES
    (NOW(), 1001, 25.5, 60.2),
    (NOW(), 1002, 26.1, 58.8),
    (NOW(), 1003, 24.8, 62.1);

-- 查询
SELECT * FROM sensor_data WHERE ts > NOW() - INTERVAL '1 hour';

-- 带条件查询
SELECT device_id, AVG(temperature)
FROM sensor_data
WHERE ts BETWEEN '2026-01-01' AND '2026-01-31'
GROUP BY device_id
HAVING AVG(temperature) > 25;
```

### 4.3 时序函数

```sql
-- 时间分桶
SELECT time_bucket('1 minute', ts) AS bucket, AVG(temperature)
FROM sensor_data
GROUP BY bucket;

-- 聚合函数
SELECT
    FIRST(temperature, ts) AS first_temp,
    LAST(temperature, ts) AS last_temp,
    MAX(temperature) AS max_temp,
    MIN(temperature) AS min_temp,
    AVG(temperature) AS avg_temp
FROM sensor_data
WHERE ts > NOW() - INTERVAL '1 day';
```

### 4.4 事务

> **⚠️ 不支持事务**：`execute_statement` 的分发中没有 `BEGIN`/`COMMIT`/`UPDATE` 分支，
> 三条语句均会报 `unsupported statement`。`tmdb-txn` crate 存在，但未接入执行引擎。

```sql
BEGIN;
INSERT INTO sensor_data VALUES (NOW(), 1001, 25.5, 60.2);
UPDATE device_meta SET status = 'active' WHERE id = 1001;
COMMIT;
```

## 6. 性能调优

### 6.1 写入优化

```sql
-- 批量插入（推荐）
INSERT INTO sensor_data VALUES
    (NOW(), 1001, 25.5, 60.2),
    (NOW(), 1002, 26.1, 58.8),
    ...;  -- 一次插入多行

-- 调整批量大小
-- 配置：storage.chunk_interval = "1 day"
-- 配置：storage.segment_target = "256MB"
```

### 6.2 查询优化

```sql
-- 使用时间范围过滤（利用 ZoneMap 裁剪）
SELECT * FROM sensor_data
WHERE ts > NOW() - INTERVAL '1 hour'  -- 只扫描最近的数据
  AND device_id = 1001;

-- 只选择需要的列（列选择优化）
SELECT ts, temperature FROM sensor_data;  -- 不读 humidity

-- 使用索引
CREATE INDEX idx_device ON sensor_data (device_id);
SELECT * FROM sensor_data WHERE device_id = 1001;
```

### 6.3 连接池优化

```toml
# 增加连接池大小
[server]
max_connections = 8192

# 读写分离
[wire.mysql]
autocommit_default = true  # 读操作自动提交
```

### 6.4 存储优化

```toml
[storage]
# 调整编码档位
default_encoding = "archive"  # 最大压缩比

# 调整 chunk 大小
chunk_interval = "7 days"  # 更大的 chunk → 更好的压缩比

# 调整段文件大小
segment_target = "512MB"  # 更大的段文件 → 更少的元数据开销
```

## 8. 迁移指南

### 8.1 从 MySQL 迁移

```bash
# 1. 导出 MySQL 数据
mysqldump -h mysql-host -u root -p --no-create-info smart_grid > data.sql

# 2. 修改连接到 tmDB
mysql -h tmdb-host -P 3306 -u tmdb < data.sql

# 3. 修改应用连接串
# jdbc:mysql://tmdb-host:3306/smart_grid
```

### 8.2 从 PostgreSQL 迁移

```bash
# 1. 导出 PG 数据
pg_dump -h pg-host -U postgres smart_grid > data.sql

# 2. 修改连接到 tmDB
psql -h tmdb-host -p 5432 -U tmdb < data.sql

# 3. 修改应用连接串
# jdbc:postgresql://tmdb-host:5432/smart_grid
```

### 8.3 从 InfluxDB 迁移

```bash
# 1. 导出 InfluxDB 数据
influx_inspect export --db mydb --out /tmp/influx.txt

# 2. 导入到 tmDB
./tmdb import --format influx --file /tmp/influx.txt --table sensor_data
```

### 8.4 从 TDengine 迁移

```bash
# 1. 导出 TDengine 数据
taosdump -D mydb -o /tmp/tdengine/

# 2. 转换并导入
./tmdb import --format tdengine --file /tmp/tdengine/ --table sensor_data
```

## 9. 常见问题

```bash

```

```bash
# 启用慢查询日志
./tmdb start --log-level debug --slow-query-threshold 1000

# 查看日志
grep "slow query" /var/log/tmdb/tmdb.log
```

```toml
[security]
sm_tls    = true    # 启用国密 TLS
sm_encrypt = true   # 启用静态加密
```

```bash
# 生成国密证书
openssl req -x509 -newkey sm2 -keyout server.key -out server.crt -days 365
```

### Q8: 如何查看集群状态？

```bash
# 查看节点状态
curl http://localhost:8080/metrics | grep tmdb_nodes

# 查看分片状态
curl http://localhost:8080/metrics | grep tmdb_shards
```
